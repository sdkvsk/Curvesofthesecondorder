/* ================================
   Utilities: Canvas coordinate system
================================ */
function makePlot(canvas){
  const ctx = canvas.getContext('2d');
  const state = {
    canvas, ctx,
    scaleUnits: 12,           // half-range in world units
    showGrid: true,
    showLabels: true,
    panX: 0, panY: 0          // world translation
  };

  function resizeToCSS(){
    const cssW = canvas.clientWidth, cssH = canvas.clientHeight;
    if (cssW && cssH){
      const dpr = window.devicePixelRatio || 1;
      const w = Math.round(cssW * dpr), h = Math.round(cssH * dpr);
      if (canvas.width !== w || canvas.height !== h){
        canvas.width = w; canvas.height = h;
      }
    }
  }

  function worldToScreen(x,y){
    const {width:w, height:h} = canvas;
    const s = Math.min(w,h)/(2*state.scaleUnits);
    // origin in center + pan
    const X = w/2 + (x - state.panX)*s;
    const Y = h/2 - (y - state.panY)*s;
    return [X,Y];
  }

  function screenToWorld(X,Y){
    const {width:w, height:h} = canvas;
    const s = Math.min(w,h)/(2*state.scaleUnits);
    const x = (X - w/2)/s + state.panX;
    const y = -(Y - h/2)/s + state.panY;
    return [x,y];
  }

  function clear(){
    resizeToCSS();
    ctx.clearRect(0,0,canvas.width,canvas.height);
  }

  function drawGrid(){
    if (!state.showGrid) return;

    const {width:w,height:h} = canvas;
    const s = Math.min(w,h)/(2*state.scaleUnits);
    const step = pickNiceStep(state.scaleUnits);
    ctx.save();
    ctx.lineWidth = 1;
    ctx.globalAlpha = 0.55;

    // minor lines
    ctx.strokeStyle = "rgba(15,23,42,0.08)";
    for (let x = Math.floor((state.panX-state.scaleUnits)/step)*step; x <= state.panX+state.scaleUnits; x += step){
      const [X0,Y0] = worldToScreen(x, state.panY-state.scaleUnits);
      const [X1,Y1] = worldToScreen(x, state.panY+state.scaleUnits);
      ctx.beginPath(); ctx.moveTo(X0,Y0); ctx.lineTo(X1,Y1); ctx.stroke();
    }
    for (let y = Math.floor((state.panY-state.scaleUnits)/step)*step; y <= state.panY+state.scaleUnits; y += step){
      const [X0,Y0] = worldToScreen(state.panX-state.scaleUnits, y);
      const [X1,Y1] = worldToScreen(state.panX+state.scaleUnits, y);
      ctx.beginPath(); ctx.moveTo(X0,Y0); ctx.lineTo(X1,Y1); ctx.stroke();
    }

    // axes
    ctx.globalAlpha = 0.9;
    ctx.strokeStyle = "rgba(15,23,42,0.20)";
    let [ax0,ay0] = worldToScreen(state.panX-state.scaleUnits, 0);
    let [ax1,ay1] = worldToScreen(state.panX+state.scaleUnits, 0);
    ctx.beginPath(); ctx.moveTo(ax0,ay0); ctx.lineTo(ax1,ay1); ctx.stroke();

    let [bx0,by0] = worldToScreen(0, state.panY-state.scaleUnits);
    let [bx1,by1] = worldToScreen(0, state.panY+state.scaleUnits);
    ctx.beginPath(); ctx.moveTo(bx0,by0); ctx.lineTo(bx1,by1); ctx.stroke();

    // labels
    if (state.showLabels){
      ctx.globalAlpha = 0.85;
      ctx.fillStyle = "rgba(71,85,105,0.85)";
      ctx.font = `${12*(window.devicePixelRatio||1)}px system-ui`;
      for (let x = Math.floor((state.panX-state.scaleUnits)/step)*step; x <= state.panX+state.scaleUnits; x += step){
        if (Math.abs(x) < 1e-9) continue;
        const [X,Y] = worldToScreen(x,0);
        ctx.fillText(String(fix(x)), X+3, Y-3);
      }
      for (let y = Math.floor((state.panY-state.scaleUnits)/step)*step; y <= state.panY+state.scaleUnits; y += step){
        if (Math.abs(y) < 1e-9) continue;
        const [X,Y] = worldToScreen(0,y);
        ctx.fillText(String(fix(y)), X+3, Y-3);
      }
    }

    ctx.restore();
  }

  function pickNiceStep(range){
    // range is half-width; choose grid step
    const raw = range/6;
    const pow = Math.pow(10, Math.floor(Math.log10(raw)));
    const m = raw/pow;
    let step;
    if (m < 1.5) step = 1*pow;
    else if (m < 3.5) step = 2*pow;
    else if (m < 7.5) step = 5*pow;
    else step = 10*pow;
    return step;
  }

  function fix(x){
    const ax = Math.abs(x);
    if (ax >= 10) return Math.round(x);
    if (ax >= 1) return Math.round(x*10)/10;
    return Math.round(x*100)/100;
  }

  function line(x1,y1,x2,y2, style="rgba(37,99,235,0.9)", width=2){
    const [X1,Y1]=worldToScreen(x1,y1), [X2,Y2]=worldToScreen(x2,y2);
    ctx.save();
    ctx.strokeStyle=style; ctx.lineWidth=width*(window.devicePixelRatio||1);
    ctx.beginPath(); ctx.moveTo(X1,Y1); ctx.lineTo(X2,Y2); ctx.stroke();
    ctx.restore();
  }

  function point(x,y, r=5, style="rgba(22,163,74,0.95)"){
    const [X,Y]=worldToScreen(x,y);
    ctx.save();
    ctx.fillStyle=style;
    ctx.beginPath(); ctx.arc(X,Y,r*(window.devicePixelRatio||1),0,Math.PI*2); ctx.fill();
    ctx.restore();
  }

  function text(x,y, t, style="rgba(51,65,85,0.92)"){
    const [X,Y]=worldToScreen(x,y);
    ctx.save();
    ctx.fillStyle=style;
    ctx.font = `${12*(window.devicePixelRatio||1)}px system-ui`;
    ctx.fillText(t, X+6, Y-6);
    ctx.restore();
  }

  function polyline(pts, style="rgba(37,99,235,0.9)", width=2){
    if (pts.length<2) return;
    ctx.save();
    ctx.strokeStyle=style;
    ctx.lineWidth=width*(window.devicePixelRatio||1);
    ctx.beginPath();
    let [X0,Y0]=worldToScreen(pts[0][0],pts[0][1]);
    ctx.moveTo(X0,Y0);
    for(let i=1;i<pts.length;i++){
      let [X,Y]=worldToScreen(pts[i][0],pts[i][1]);
      ctx.lineTo(X,Y);
    }
    ctx.stroke();
    ctx.restore();
  }

  return {state, clear, drawGrid, worldToScreen, screenToWorld, line, point, text, polyline, resizeToCSS};
}

function drawCircle(plot, R){
  const {ctx, canvas} = plot.state;
  plot.clear(); plot.drawGrid();

  const pts=[];
  for(let i=0;i<=360;i++){
    const t=i*Math.PI/180;
    pts.push([R*Math.cos(t), R*Math.sin(t)]);
  }
  plot.polyline(pts, "rgba(37,99,235,0.95)", 2.2);
  plot.point(R,0,4); plot.point(0,R,4);
  plot.text(R,0, `R=${R.toFixed(2)}`);
}

function drawEllipse(plot, a, b, showFoci){
  plot.clear(); plot.drawGrid();

  // ensure a>=b for c
  let swapped=false;
  let A=a, B=b;
  if (B > A){ [A,B]=[B,A]; swapped=true; }
  const c = Math.sqrt(Math.max(0, A*A - B*B));

  const pts=[];
  for(let i=0;i<=360;i++){
    const t=i*Math.PI/180;
    pts.push([a*Math.cos(t), b*Math.sin(t)]);
  }
  plot.polyline(pts, "rgba(37,99,235,0.95)", 2.2);

  // axes
  plot.line(-a,0,a,0,"rgba(15,23,42,0.28)",1.6);
  plot.line(0,-b,0,b,"rgba(15,23,42,0.28)",1.6);
  plot.text(a,0,"a"); plot.text(0,b,"b");

  if (showFoci){
    // foci depend on major axis direction; in our drawing major axis is x if a>=b else y
    if (!swapped){
      plot.point(c,0,4,"rgba(22,163,74,0.95)");
      plot.point(-c,0,4,"rgba(22,163,74,0.95)");
      plot.text(c,0,"F₁"); plot.text(-c,0,"F₂");
    } else {
      plot.point(0,c,4,"rgba(22,163,74,0.95)");
      plot.point(0,-c,4,"rgba(22,163,74,0.95)");
      plot.text(0,c,"F₁"); plot.text(0,-c,"F₂");
    }
  }
  return {c, majorA:A, minorB:B, swapped};
}

function drawHyperbola(plot, a, b, showAsy, showFoci){
  plot.clear(); plot.drawGrid();

  // branches via cosh/sinh
  const tMax = 2.2;
  const pts1=[], pts2=[];
  for (let i=0;i<=240;i++){
    const t = -tMax + (2*tMax)*i/240;
    const x = a*Math.cosh(t);
    const y = b*Math.sinh(t);
    pts1.push([x,y]);
    pts2.push([-x,y]);
  }
  plot.polyline(pts1,"rgba(37,99,235,0.95)",2.2);
  plot.polyline(pts2,"rgba(37,99,235,0.95)",2.2);

  if (showAsy){
    const m = b/a;
    const L = plot.state.scaleUnits*1.2;
    plot.line(-L, -m*L, L, m*L, "rgba(15,23,42,0.24)",1.6);
    plot.line(-L, m*L, L, -m*L, "rgba(15,23,42,0.24)",1.6);
  }

  if (showFoci){
    const c = Math.sqrt(a*a + b*b);
    plot.point(c,0,4,"rgba(22,163,74,0.95)");
    plot.point(-c,0,4,"rgba(22,163,74,0.95)");
    plot.text(c,0,"F₁"); plot.text(-c,0,"F₂");
  }

  // vertices
  plot.point(a,0,4,"rgba(245,158,11,0.95)");
  plot.point(-a,0,4,"rgba(245,158,11,0.95)");
  plot.text(a,0,"V"); plot.text(-a,0,"V");
  return {c: Math.sqrt(a*a + b*b)};
}

function drawParabola(plot, p, orient="right", showFD=true){
  plot.clear(); plot.drawGrid();
  // canonical: y^2 = 2px (right). For other orientations: rotate/reflect
  const pts=[];
  const xMin = -plot.state.scaleUnits, xMax = plot.state.scaleUnits;
  const samples = 300;

  function mapXY(x,y){
    switch(orient){
      case "right": return [x,y];
      case "left":  return [-x,y];
      case "up":    return [y,x];  // swap -> x^2 = 2py
      case "down":  return [y,-x];
    }
    return [x,y];
  }

  // build in right orientation then map
  for(let i=0;i<=samples;i++){
    const x = (0 + (xMax-0)*i/samples);
    const y = Math.sqrt(Math.max(0, 2*p*x));
    const [X1,Y1] = mapXY(x, y);
    pts.push([X1,Y1]);
  }
  const pts2=[];
  for(let i=samples;i>=0;i--){
    const x = (0 + (xMax-0)*i/samples);
    const y = -Math.sqrt(Math.max(0, 2*p*x));
    const [X1,Y1] = mapXY(x, y);
    pts2.push([X1,Y1]);
  }
  plot.polyline(pts.concat(pts2), "rgba(37,99,235,0.95)", 2.2);

  if (showFD){
    // focus and directrix for right orientation:
    // focus (p/2,0), directrix x = -p/2
    let fx = p/2, fy = 0;
    let dir = {x:-p/2, y:null}; // vertical line x = -p/2

    // map for other orientations
    function mapPoint(x,y){ return mapXY(x,y); }
    function mapLineX(constX){
      // x=const vertical line mapped under our crude transforms above
      // For right/left: stays vertical
      // For up/down: becomes horizontal y=const
      if (orient==="right") return {type:"x", v:constX};
      if (orient==="left")  return {type:"x", v:-constX};
      if (orient==="up")    return {type:"y", v:constX};
      if (orient==="down")  return {type:"y", v:-constX};
      return {type:"x", v:constX};
    }

    const [fX,fY] = mapPoint(fx,fy);
    plot.point(fX,fY,4,"rgba(22,163,74,0.95)");
    plot.text(fX,fY,"F");

    const L = plot.state.scaleUnits*1.2;
    const lineSpec = mapLineX(-p/2);
    if (lineSpec.type==="x"){
      plot.line(lineSpec.v, -L, lineSpec.v, L, "rgba(15,23,42,0.24)", 1.6);
      plot.text(lineSpec.v, 0, "директриса");
    } else {
      plot.line(-L, lineSpec.v, L, lineSpec.v, "rgba(15,23,42,0.24)", 1.6);
      plot.text(0, lineSpec.v, "директриса");
    }
  }
}

/* ================================
   Implicit plotting (marching squares)
================================ */
function drawImplicit(plot, f, opts={}){
  const {ctx, canvas} = plot.state;
  plot.clear(); plot.drawGrid();

  const w = canvas.width, h = canvas.height;
  const stepPx = Math.max(6, Math.round(10*(window.devicePixelRatio||1)));
  const cols = Math.floor(w/stepPx);
  const rows = Math.floor(h/stepPx);

  // Sample scalar field on grid
  const values = new Array((cols+1)*(rows+1));
  for(let j=0;j<=rows;j++){
    for(let i=0;i<=cols;i++){
      const X = i*stepPx;
      const Y = j*stepPx;
      const [x,y] = plot.screenToWorld(X,Y);
      values[j*(cols+1)+i] = f(x,y);
    }
  }

  // Marching squares for isovalue 0
  ctx.save();
  ctx.strokeStyle = "rgba(37,99,235,0.95)";
  ctx.lineWidth = 2.2*(window.devicePixelRatio||1);
  ctx.globalAlpha = 0.95;

  function interp(p1, p2, v1, v2){
    const t = v1/(v1 - v2); // value=0
    return [ p1[0] + t*(p2[0]-p1[0]), p1[1] + t*(p2[1]-p1[1]) ];
  }

  for(let j=0;j<rows;j++){
    for(let i=0;i<cols;i++){
      const idx = j*(cols+1)+i;
      const v0 = values[idx];
      const v1 = values[idx+1];
      const v2 = values[idx+(cols+1)+1];
      const v3 = values[idx+(cols+1)];

      // corners in screen px
      const p0 = [i*stepPx, j*stepPx];
      const p1 = [(i+1)*stepPx, j*stepPx];
      const p2 = [(i+1)*stepPx, (j+1)*stepPx];
      const p3 = [i*stepPx, (j+1)*stepPx];

      let c = 0;
      if (v0 >= 0) c |= 1;
      if (v1 >= 0) c |= 2;
      if (v2 >= 0) c |= 4;
      if (v3 >= 0) c |= 8;

      // edges: e0 p0-p1, e1 p1-p2, e2 p2-p3, e3 p3-p0
      const edges = [];
      switch(c){
        case 0: case 15: break;
        case 1: case 14:
          edges.push([interp(p3,p0,v3,v0), interp(p0,p1,v0,v1)]); break;
        case 2: case 13:
          edges.push([interp(p0,p1,v0,v1), interp(p1,p2,v1,v2)]); break;
        case 3: case 12:
          edges.push([interp(p3,p0,v3,v0), interp(p1,p2,v1,v2)]); break;
        case 4: case 11:
          edges.push([interp(p1,p2,v1,v2), interp(p2,p3,v2,v3)]); break;
        case 5:
          edges.push([interp(p3,p0,v3,v0), interp(p0,p1,v0,v1)]);
          edges.push([interp(p1,p2,v1,v2), interp(p2,p3,v2,v3)]);
          break;
        case 6: case 9:
          edges.push([interp(p0,p1,v0,v1), interp(p2,p3,v2,v3)]); break;
        case 7: case 8:
          edges.push([interp(p3,p0,v3,v0), interp(p2,p3,v2,v3)]); break;
        case 10:
          edges.push([interp(p0,p1,v0,v1), interp(p1,p2,v1,v2)]);
          edges.push([interp(p2,p3,v2,v3), interp(p3,p0,v3,v0)]);
          break;
      }

      for (const seg of edges){
        ctx.beginPath();
        ctx.moveTo(seg[0][0], seg[0][1]);
        ctx.lineTo(seg[1][0], seg[1][1]);
        ctx.stroke();
      }
    }
  }
  ctx.restore();
}

/* ================================
   Conic classification & canonical hints (numeric)
   We compute invariants and (approx) rotation angle to remove xy term.
================================ */
function classifyConic(A,B,C,D,E,F){
  // invariants
  const delta = B*B - 4*A*C;
  // determinant of the 3x3 (degeneracy test)
  const det =
    A*(C*F - (E*E)/4) -
    (B/2)*((B/2)*F - (D*E)/4) +
    (D/2)*((B/2)*(E/2) - C*(D/2));
  // robust det (explicit formula)
  const Det = A*(C*F - (E*E)/4) - (B/2)*((B/2)*F - (D*E)/4) + (D/2)*((B*E)/4 - C*(D/2));

  let type;
  if (Math.abs(delta) < 1e-9) type = "Парабола (або вироджений випадок)";
  else if (delta < 0) type = "Еліпс / коло (або уявний еліпс)";
  else type = "Гіпербола";

  // rotation to eliminate xy:
  let theta = 0;
  if (Math.abs(B) > 1e-12 && Math.abs(A-C) > 1e-12){
    theta = 0.5 * Math.atan2(B, A - C);
  } else if (Math.abs(B) > 1e-12 && Math.abs(A-C) <= 1e-12){
    theta = Math.PI/4; // approximate
  }

  return {delta, type, theta, det: Det};
}

function rotateCoeffs(A,B,C,D,E,F, theta){
  // rotate coordinates: x = X cos - Y sin, y = X sin + Y cos
  const c = Math.cos(theta), s = Math.sin(theta);

  // Transform quadratic form
  // Using formulas for rotated conic coefficients:
  const A2 = A*c*c + B*c*s + C*s*s;
  const B2 = 2*(C-A)*s*c + B*(c*c - s*s);
  const C2 = A*s*s - B*s*c + C*c*c;

  const D2 = D*c + E*s;
  const E2 = -D*s + E*c;

  return {A:A2, B:B2, C:C2, D:D2, E:E2, F};
}

// dragging chord points
(function setupChordDrag(){
  const cv = document.getElementById('cv-elements');
  let dragging = -1;

  function hitTest(mx,my){
    const plot = plots.elements;
    const [x,y] = plot.screenToWorld(mx,my);
    const p0 = ellipsePoint(elState.a, elState.b, chordPts[0].t);
    const p1 = ellipsePoint(elState.a, elState.b, chordPts[1].t);
    const d0 = (x-p0[0])**2 + (y-p0[1])**2;
    const d1 = (x-p1[0])**2 + (y-p1[1])**2;
    const thr = (0.6)**2;
    if (d0 < thr) return 0;
    if (d1 < thr) return 1;
    return -1;
  }

  cv.addEventListener('pointerdown', (e)=>{
    const rect = cv.getBoundingClientRect();
    const mx = (e.clientX-rect.left)*(window.devicePixelRatio||1);
    const my = (e.clientY-rect.top)*(window.devicePixelRatio||1);
    dragging = hitTest(mx,my);
    if (dragging>=0){
      cv.setPointerCapture(e.pointerId);
    }
  });

  cv.addEventListener('pointermove', (e)=>{
    if (dragging<0) return;
    const rect = cv.getBoundingClientRect();
    const mx = (e.clientX-rect.left)*(window.devicePixelRatio||1);
    const my = (e.clientY-rect.top)*(window.devicePixelRatio||1);
    const plot = plots.elements;
    const [x,y] = plot.screenToWorld(mx,my);
    // project to ellipse parameter angle by atan2(y/b, x/a)
    const t = Math.atan2(y/(elState.b||1e-9), x/(elState.a||1e-9));
    chordPts[dragging].t = t;
    redrawAll();
  });

  cv.addEventListener('pointerup', (e)=>{ dragging=-1; });
  cv.addEventListener('pointercancel', ()=>{ dragging=-1; });
})();

// pan on general canvas
(function setupGeneralPan(){
  const cv = document.getElementById('cv-general');
  let dragging=false;
  let last=null;

  cv.addEventListener('pointerdown', (e)=>{
    if (!gState.panEnabled) return;
    dragging=true;
    const rect=cv.getBoundingClientRect();
    last=[(e.clientX-rect.left)*(window.devicePixelRatio||1),(e.clientY-rect.top)*(window.devicePixelRatio||1)];
    cv.setPointerCapture(e.pointerId);
  });
  cv.addEventListener('pointermove', (e)=>{
    if (!dragging) return;
    const rect=cv.getBoundingClientRect();
    const cur=[(e.clientX-rect.left)*(window.devicePixelRatio||1),(e.clientY-rect.top)*(window.devicePixelRatio||1)];
    const plot=plots.general;
    // convert screen delta to world delta
    const [x0,y0]=plot.screenToWorld(last[0], last[1]);
    const [x1,y1]=plot.screenToWorld(cur[0], cur[1]);
    const dx=x0-x1, dy=y0-y1;
    gState.panX += dx;
    gState.panY += dy;
    last=cur;
    redrawAll();
  });
  cv.addEventListener('pointerup', ()=>{dragging=false;});
  cv.addEventListener('pointercancel', ()=>{dragging=false;});
})();

/* ================================
   Redraw
================================ */
function redrawAll(){
  // Circle
  {
    const p = plots.circle;
    p.state.scaleUnits = circleState.scale;
    p.state.showGrid = circleState.grid;
    p.state.showLabels = circleState.labels;
    p.state.panX = 0; p.state.panY = 0;
    drawCircle(p, circleState.R);
  }

  // Ellipse
  {
    const p = plots.ellipse;
    p.state.scaleUnits = ellState.scale;
    p.state.showGrid = true;
    p.state.showLabels = true;
    p.state.panX = 0; p.state.panY = 0;
    const info = drawEllipse(p, ellState.a, ellState.b, ellState.foci);
    document.getElementById('ellC').textContent = info.c.toFixed(3);
    document.getElementById('ellArea').textContent = (Math.PI*ellState.a*ellState.b).toFixed(3);
  }

  // Hyperbola
  {
    const p = plots.hyperbola;
    p.state.scaleUnits = hypState.scale;
    p.state.showGrid = true;
    p.state.showLabels = true;
    p.state.panX = 0; p.state.panY = 0;
    const info = drawHyperbola(p, hypState.a, hypState.b, hypState.asy, hypState.foci);
    document.getElementById('hypC').textContent = info.c.toFixed(3);
  }

  // Parabola
  {
    const p = plots.parabola;
    p.state.scaleUnits = parState.scale;
    p.state.showGrid = true;
    p.state.showLabels = true;
    p.state.panX = 0; p.state.panY = 0;
    drawParabola(p, parState.p, parState.orient, parState.showFD);
  }

  // Elements
  {
    const p = plots.elements;
    p.state.scaleUnits = elState.scale;
    p.state.showGrid = true;
    p.state.showLabels = true;
    p.state.panX = 0; p.state.panY = 0;

    p.clear(); p.drawGrid();
    // ellipse
    const pts=[];
    for(let i=0;i<=360;i++){
      const t=i*Math.PI/180;
      pts.push([elState.a*Math.cos(t), elState.b*Math.sin(t)]);
    }
    p.polyline(pts, "rgba(37,99,235,0.95)", 2.2);

    if (elState.axes){
      p.line(-elState.a,0,elState.a,0,"rgba(15,23,42,0.28)",1.6);
      p.line(0,-elState.b,0,elState.b,"rgba(15,23,42,0.28)",1.6);
      p.text(elState.a,0,"a"); p.text(0,elState.b,"b");
    }

    const P0 = ellipsePoint(elState.a, elState.b, chordPts[0].t);
    const P1 = ellipsePoint(elState.a, elState.b, chordPts[1].t);
    p.point(P0[0],P0[1],4,"rgba(22,163,74,0.95)");
    p.point(P1[0],P1[1],4,"rgba(22,163,74,0.95)");

    if (elState.chord){
      p.line(P0[0],P0[1],P1[0],P1[1],"rgba(13,148,136,0.55)",2);
      p.text((P0[0]+P1[0])/2,(P0[1]+P1[1])/2,"хорда");
    }

    if (elState.diameter){
      // diameter parallel to chord passing through origin: same direction vector
      const vx = P1[0]-P0[0], vy = P1[1]-P0[1];
      const L = p.state.scaleUnits*1.1;
      const norm = Math.hypot(vx,vy)||1;
      const ux = vx/norm, uy = vy/norm;
      p.line(-L*ux, -L*uy, L*ux, L*uy, "rgba(255,209,102,0.55)", 2);
      p.text(L*ux, L*uy, "діаметр");
    }

    const chordLen = Math.hypot(P1[0]-P0[0], P1[1]-P0[1]);
    document.getElementById('chordLen').textContent = chordLen.toFixed(3);
    document.getElementById('elArea').textContent = (Math.PI*elState.a*elState.b).toFixed(3);
  }

  // Optics
  {
    const p = plots.optics;
    p.state.scaleUnits = opState.scale;
    p.state.showGrid = true;
    p.state.showLabels = true;
    p.state.panX = 0; p.state.panY = 0;

    p.clear(); p.drawGrid();

    const ctx = p.state.ctx;
    // optional "double cone" schematic in corner
    if (opState.showCone){
      ctx.save();
      ctx.globalAlpha = 0.9;
      ctx.fillStyle = "rgba(15,23,42,0.08)";
      ctx.strokeStyle = "rgba(15,23,42,0.24)";
      ctx.lineWidth = 1.6*(window.devicePixelRatio||1);

      const pad = 18*(window.devicePixelRatio||1);
      const W = 170*(window.devicePixelRatio||1);
      const H = 140*(window.devicePixelRatio||1);
      const x0 = pad, y0 = pad;

      ctx.beginPath();
      ctx.roundRect(x0, y0, W, H, 16*(window.devicePixelRatio||1));
      ctx.fill(); ctx.stroke();

      // draw cone
      const cx = x0 + W*0.5, cy = y0 + H*0.55;
      ctx.beginPath();
      ctx.moveTo(cx, y0+12*(window.devicePixelRatio||1));
      ctx.lineTo(x0+W*0.25, cy);
      ctx.lineTo(cx, y0+H-12*(window.devicePixelRatio||1));
      ctx.lineTo(x0+W*0.75, cy);
      ctx.closePath();
      ctx.stroke();

      // plane
      ctx.strokeStyle = "rgba(37,99,235,0.55)";
      ctx.beginPath();
      ctx.moveTo(x0+W*0.15, cy);
      ctx.lineTo(x0+W*0.85, cy);
      ctx.stroke();

      ctx.fillStyle="rgba(51,65,85,0.78)";
      ctx.font = `${11*(window.devicePixelRatio||1)}px system-ui`;
      ctx.fillText("Подвійний конус", x0+12*(window.devicePixelRatio||1), y0+H-12*(window.devicePixelRatio||1));
      ctx.restore();
    }

    // rays
    const raysN = opState.rays;
    if (opState.mode==="parabola"){
      // draw parabola opening right
      drawParabola(p, opState.p, "right", true);
      // overlay rays: parallel to x-axis hitting parabola, then reflected to focus (demo)
      const focus = [opState.p/2, 0];

      for (let i=0;i<raysN;i++){
        const y = -opState.p*0.9 + (2*opState.p*0.9)*i/(raysN-1);
        // choose intersection point approximately: y^2 = 2px -> x = y^2/(2p)
        const xHit = (y*y)/(2*opState.p);
        // incoming segment
        p.line(-p.state.scaleUnits, y, xHit, y, "rgba(15,23,42,0.18)", 1.6);
        // "reflected" segment to focus (demo line)
        p.line(xHit, y, focus[0], focus[1], "rgba(13,148,136,0.35)", 1.6);
      }
      p.point(focus[0],focus[1],4,"rgba(22,163,74,0.95)");
      p.text(focus[0],focus[1],"F");
    } else {
      // ellipse mode
      const a = opState.a, b = opState.b;
      const info = drawEllipse(p, a, b, true);
      // foci positions (assuming major axis along x when a>=b)
      let F1, F2;
      if (a>=b){
        const c = Math.sqrt(Math.max(0, a*a - b*b));
        F1=[-c,0]; F2=[c,0];
      } else {
        const c = Math.sqrt(Math.max(0, b*b - a*a));
        F1=[0,-c]; F2=[0,c];
      }
      // pick points on ellipse and show segments F1->P->F2 (optical property concept)
      for (let i=0;i<raysN;i++){
        const t = (i/(raysN))*Math.PI*2;
        const P=[a*Math.cos(t), b*Math.sin(t)];
        p.line(F1[0],F1[1],P[0],P[1],"rgba(15,23,42,0.24)",1.6);
        p.line(P[0],P[1],F2[0],F2[1],"rgba(13,148,136,0.30)",1.6);
      }
      p.point(F1[0],F1[1],4,"rgba(22,163,74,0.95)");
      p.point(F2[0],F2[1],4,"rgba(22,163,74,0.95)");
      p.text(F1[0],F1[1],"F₁"); p.text(F2[0],F2[1],"F₂");
    }
  }

  // General equation
  {
    const p = plots.general;
    p.state.scaleUnits = gState.scale;
    p.state.showGrid = true;
    p.state.showLabels = true;
    p.state.panX = gState.panX;
    p.state.panY = gState.panY;

    const A=gState.A, B=gState.B, C=gState.C, D=gState.D, E=gState.E, F=gState.F;

    const cls = classifyConic(A,B,C,D,E,F);
    const thetaDeg = cls.theta*180/Math.PI;

    // Use implicit function f(x,y)
    const f = (x,y) => A*x*x + B*x*y + C*y*y + D*x + E*y + F;
    drawImplicit(p, f);

    // show a "rotation suggestion"
    const rot = rotateCoeffs(A,B,C,D,E,F, cls.theta);
    const B2 = rot.B;

    const info = `
      Тип за дискримінантом: <b>${cls.type}</b><br/>
      \\(\\Delta=B^2-4AC=\\) <b>${cls.delta.toFixed(3)}</b><br/>
      Кут повороту для усунення \\(xy\\): \\(\\theta\\approx\\) <b>${thetaDeg.toFixed(2)}°</b><br/>
      Після повороту: \\(B'\\approx\\) <b>${B2.toFixed(4)}</b> (чим ближче до 0 — тим краще)<br/>
      Рівняння: \\(${A.toFixed(3)}x^2 + ${B.toFixed(3)}xy + ${C.toFixed(3)}y^2 + ${D.toFixed(3)}x + ${E.toFixed(3)}y + ${F.toFixed(3)}=0\\)
    `;
    const el = document.getElementById('genInfo');
    el.innerHTML = info;
    // re-typeset math
    if (window.MathJax && window.MathJax.typesetPromise){
      MathJax.typesetPromise([el]).catch(()=>{});
    }
  }
}
